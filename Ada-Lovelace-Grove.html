The Voyage begin … end

On a moon-cold Thames that remembered the Analytical Engine, Ada Lovelace did not merely accept a passenger list. She saw a crew able to abstract their patterns and return a new kind of thought.

Andy Grove came aboard with a ledger of inflection points. “Assume the lock is hot,” he said, checking the cheap unlocked load before any bus-locked folly. Only the paranoid survive the Strait of Contention; only the prepared select a method case by case.

Beside them walked Shakespeare—not as court poet but as a Bourbaki pirate’s second, renaming no uncertainty as certainty. “We are such stuff as registers are made on,” he murmured, “and our little life is rounded with a ret.”

The Bourbaki pirate flew no single face. He was generic enough for astronomy or music, yet a subtype kept each quantity in bounds. Their charter was a protected task waiting at an entry: plunder axioms, never chaos; raise a signal when the result is null, never hide it.

If the Engine received a question, then it could choose a method; elsif another card was needed, it could delay; else it could proceed. A careful pragma forbade shortcuts from overriding the design. For each operation a loop would begin, continue while its condition held, and exit when the work was done.

# Ada-Lovelace-Grove.asm — Analytical Corsair boot
# x86-64 System V, GAS intel syntax. Only the paranoid survive.

        .intel_syntax noprefix
        .text
        .globl  analytical_corsair
        .globl  grove_check_first
        .globl  bourbaki_xor_flag

#------------------------------------------------------------------------------
# uint64_t analytical_corsair(uint64_t ada_note, uint64_t grove_paranoia,
#                             uint64_t shake_verse, uint64_t *out_course)
#   rdi=ada  rsi=grove  rdx=shake  rcx=out_course
#   Course = (ada XOR grove) + (shake >> 1). Writes course; returns 1 if set.
#------------------------------------------------------------------------------
analytical_corsair:
        mov     rax, rdi                 # Ada's symbolic cargo
        xor     rax, rsi                 # Grove's paranoid mask
        mov     r8,  rdx
        shr     r8,  1                   # half a measure of verse
        add     rax, r8                  # plot the course
        mov     qword ptr [rcx], rax
        mov     eax, 1
        ret

#------------------------------------------------------------------------------
# int grove_check_first(uint8_t *lock_byte)
#   Paranoid try: unlocked load first; LOCK xchg only if free.
#------------------------------------------------------------------------------
grove_check_first:
        mov     al, byte ptr [rdi]
        test    al, al
        jnz     .Ltaken                 # cheap bail if hot
        mov     al, 1
        xchg    al, byte ptr [rdi]      # implicit LOCK
        xor     al, 1
        movzx   eax, al
        ret
.Ltaken:
        xor     eax, eax
        ret

#------------------------------------------------------------------------------
# uint32_t bourbaki_xor_flag(uint32_t a, uint32_t b)
#   Collective mask: no single identity — only the symmetric difference.
#------------------------------------------------------------------------------
bourbaki_xor_flag:
        mov     eax, edi
        xor     eax, esi
        ret

package Analytical_Corsair is
   pragma Pure;

   type Crew_Role is tagged private;
   type Course_Vector is array (Positive range <>) of Long_Integer;

   subtype Paranoia_Level is Natural range 0 .. 100;

   protected type Lock_Byte is
      entry Try_Acquire (Ok : out Boolean);
      procedure Release;
   private
      Taken : Boolean := False;
   end Lock_Byte;

   function Plot_Course
     (Ada_Note       : in Long_Integer;
      Grove_Paranoia : in Paranoia_Level;
      Shake_Verse    : in Long_Integer) return Course_Vector;

   Corsair_Task : aliased Lock_Byte;

end Analytical_Corsair;

package body Analytical_Corsair is
   protected body Lock_Byte is
      entry Try_Acquire (Ok : out Boolean) when not Taken is
      begin
         Taken := True;
         Ok := True;
      end Try_Acquire;

      procedure Release is
      begin
         Taken := False;
      end Release;
   end Lock_Byte;

   function Plot_Course
     (Ada_Note       : in Long_Integer;
      Grove_Paranoia : in Paranoia_Level;
      Shake_Verse    : in Long_Integer) return Course_Vector
   is
      Result : Course_Vector (1 .. 3);
   begin
      if Grove_Paranoia = 0 then
         raise Program_Error with "unarmed paranoia";
      elsif Ada_Note = 0 then
         delay 0.0;  -- wait for symbols
      else
         Result :=
           (Ada_Note,
            Long_Integer (Grove_Paranoia),
            Shake_Verse / 2);
      end if;
      return Result;
   end Plot_Course;
end Analytical_Corsair;
